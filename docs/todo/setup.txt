ALEXANDRIA-NEXUS SETUP PLAN
=============================

Rebuild philosophie-bib from scratch using the improved hexforge SDK.
Same data model, same endpoints, dramatically less boilerplate.


CONTEXT
-------
philosophie-bib: 15K lines across 99 files, 3 workspace crates.
Most of that is boilerplate hexforge now eliminates:
  - 8 manual DataSource impls (~1200 lines) → PgQuery trait (~240 lines)
  - 8 manual PgEntity impls (~400 lines) → #[derive(Entity)] (~80 lines)
  - Manual error mapping → hexforge::map_sqlx_error
  - Manual pool setup → hexforge::DatabaseConfig
  - Manual server startup → hexforge::serve()
  - No OpenAPI → serve_openapi() + #[derive(ToSchema)]
  - Wide open CORS → CorsConfig::from_env()
  - N+1 queries → find_by_ids()
  - 46-column SELECT * everywhere → #[derive(Projection)] views

Target: same functionality, ~60% less code, fully typed OpenAPI spec.

PHASE 3: ENTITIES WITH #[derive(Entity)]
-------------------------------------------

Replace all manual PgEntity impls + RawEntity types with:

  #[derive(Entity, FromRow, Clone, Debug, Serialize, Deserialize, ToSchema)]
  #[entity(table = "organizations")]
  struct Author {
      #[entity(id)]
      id: i64,
      author_key: String,
      given_name_latex: Option<String>,
      given_name_unicode: Option<String>,
      given_name_simplified: Option<String>,
      family_name_latex: Option<String>,
      family_name_unicode: Option<String>,
      family_name_simplified: Option<String>,
      // ... etc
  }

No manual SQL. No RawAuthor intermediate type. No From<RawAuthor>
conversion. The Entity + FromRow derives handle everything.

Do this for all 8 entity types:
  Author, BibItem, Journal, Keyword, Publisher, Institution, School, Series

Plus api_keys (for auth).


PHASE 4: QUERIES WITH PgQuery
-------------------------------

Replace manual DataSource impls with PgQuery:

  #[derive(Default, Deserialize)]
  struct AuthorQuery {
      family_name: Option<String>,
      search_term: Option<String>,
  }

  impl PgQuery for AuthorQuery {
      fn build_conditions(&self, mut idx: usize) -> (Vec<String>, usize) {
          let mut conds = vec![];
          if self.family_name.is_some() {
              conds.push(format!("family_name_simplified ILIKE '%' || ${idx} || '%'"));
              idx += 1;
          }
          (conds, idx)
      }
      fn bind(&self, binder: &mut ParamBinder) {
          if let Some(ref name) = self.family_name {
              binder.add(name.clone());
          }
      }
  }

No manual find_all, count, stream, insert, update, delete.
PostgresDataSource<Author, AuthorQuery> handles all of it.

Custom lookups use find_one_where:
  ds.find_one_where("author_key = $1", |q| q.bind(key)).await

Register as views:

PHASE 6: API COMPOSITION
--------------------------

  pub fn build_app(pool: DatabasePool) -> axum::Router {
      let state = AppState::new(pool.clone());
      let validator = ApiKeyValidator::new(pool);

      Api::new()
          .with_auth(validator)

          // 8 CRUD resources
          .crud_with_meta(
              CrudResourceConfig::new("/api/v1/authors", state.author_ds.clone())
                  .create_validator(validate_create_author)
                  .update_validator(validate_update_author)
                  .create_transform(create_author_transform)
                  .update_transform(update_author_transform)
                  .read_permission(Permission::Read)
                  .create_permission(Permission::Write)
                  .update_permission(Permission::Write)
                  .delete_permission(Permission::Admin)
                  .view("summary", build_projection_view::<_, _, AuthorSummary>(..)),
              CrudOpenApiMeta::new("/api/v1/authors", "Authors")
                  .with_schemas::<Author, CreateAuthor, UpdateAuthor>()
                  .description("Author management"),
          )
          // ... repeat for all 8 entities

          // Junction tables
          .resource(Resource::<AppState>::new("/api/v1/bibitems")
              .get("/{id}/authors", get_bibitem_authors)
              .post("/{id}/authors", add_author_to_bibitem)
              .delete("/{id}/authors/{author_id}", remove_author_from_bibitem)
              .put("/{id}/authors", replace_bibitem_authors)
              .with_state(state.clone()))

          // Search (use .search_handler on bibitems CRUD)
          // Or keep as separate resource for flexibility

          // Admin
          .resource(Resource::<AppState>::new("/api/v1/admin")
              .require_permission(Permission::Admin)
              .post("/import", import_bibitems)
              .post("/import/file", import_file)
              .get("/export", export_bibitems)
              .with_state(state.clone()))

          // OpenAPI
          .serve_openapi("/docs/openapi.json",
              OpenApiConfig::new("Alexandria Nexus", "0.1.0")
                  .description("Bibliography and knowledge engine for Philosophie.ch"))

          // CORS
          .with_cors(CorsConfig::from_env("ALLOWED_ORIGINS", "http://localhost:3000"))

          .build()
  }


PHASE 7: MAIN.RS
------------------

  #[tokio::main]
  async fn main() -> Result<(), Box<dyn std::error::Error>> {
      tracing_subscriber::registry()
          .with(EnvFilter::try_from_default_env()
              .unwrap_or_else(|_| "alexandria_nexus=debug,hexforge=debug".into()))
          .with(tracing_subscriber::fmt::layer())
          .init();

      let database_url = std::env::var("DATABASE_URL")
          .unwrap_or_else(|_| "postgres://bib:password@localhost:5432/bibliography".into());

      let pool = DatabaseConfig::new(&database_url)
          .max_connections(50)
          .min_connections(5)
          .acquire_timeout(Duration::from_secs(10))
          .max_lifetime(Duration::from_secs(1800))
          .idle_timeout(Duration::from_secs(600))
          .stream_timeout(Duration::from_secs(300))
          .transaction_timeout(Duration::from_secs(30))
          .connect()
          .await?;

      sqlx::migrate!("./migrations").run(pool.pool()).await?;

      // Seed API key
      if let Ok(seed_key) = std::env::var("SEED_API_KEY") {
          // ... hash + insert
      }

      let app = build_app(pool);
      let addr = format!("{}:{}",
          std::env::var("HOST").unwrap_or_else(|_| "0.0.0.0".into()),
          std::env::var("PORT").unwrap_or_else(|_| "8080".into()));

      hexforge::serve(app, &addr).await?;
      Ok(())
  }
